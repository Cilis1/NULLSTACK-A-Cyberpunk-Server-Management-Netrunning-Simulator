cmake_minimum_required(VERSION 3.25)
project(NULLSTACK VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)
add_library(nullstack_core STATIC src/core/simulation.cpp src/core/replay_log.cpp src/core/game.cpp)
target_include_directories(nullstack_core PUBLIC src)
target_compile_features(nullstack_core PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(nullstack_core PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(nullstack_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
# Persistence is a separate target: the simulation remains independent of SQLite.
find_package(SQLite3 QUIET)
if(NOT SQLite3_FOUND)
  include(FetchContent)
  FetchContent_Declare(sqlite_amalgamation
    URL https://codeload.github.com/gittiver/sqlite3-amalgamation/tar.gz/0d6c232267e6d90f84681b2c9c151237124de404
    URL_HASH SHA256=19674157c5c1616a6183ec5302093c1fa69451228c56c87c83feebbcbda6e30e
    SOURCE_SUBDIR not-a-cmake-project)
  FetchContent_MakeAvailable(sqlite_amalgamation)
  add_library(nullstack_sqlite STATIC ${sqlite_amalgamation_SOURCE_DIR}/src/sqlite3/sqlite3.c)
  target_include_directories(nullstack_sqlite PUBLIC ${sqlite_amalgamation_SOURCE_DIR}/src/sqlite3)
  if(UNIX)
    find_package(Threads REQUIRED)
    target_link_libraries(nullstack_sqlite PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
  endif()
  add_library(SQLite::SQLite3 ALIAS nullstack_sqlite)
endif()
add_library(nullstack_save STATIC src/save/sqlite_save.cpp)
target_link_libraries(nullstack_save PUBLIC nullstack_core PRIVATE SQLite::SQLite3)
target_compile_features(nullstack_save PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(nullstack_save PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(nullstack_save PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_executable(Nullstack src/game/main.cpp)
target_link_libraries(Nullstack PRIVATE nullstack_core)

add_executable(nullstack_headless src/headless/main.cpp)
target_link_libraries(nullstack_headless PRIVATE nullstack_save)
if(BUILD_TESTING)
  add_executable(nullstack_game_tests tests/game_tests.cpp)
  target_link_libraries(nullstack_game_tests PRIVATE nullstack_core)
  add_test(NAME game COMMAND nullstack_game_tests)
  add_test(NAME game_cli COMMAND ${CMAKE_COMMAND} -DGAME=$<TARGET_FILE:Nullstack> -P ${CMAKE_SOURCE_DIR}/tests/integration/game_cli.cmake)
  add_executable(nullstack_tests tests/core_tests.cpp)
  target_link_libraries(nullstack_tests PRIVATE nullstack_core)
  add_test(NAME core COMMAND nullstack_tests)
  add_executable(nullstack_replay_tests tests/replay_tests.cpp)
  target_link_libraries(nullstack_replay_tests PRIVATE nullstack_core)
  add_test(NAME replay COMMAND nullstack_replay_tests)
  add_executable(nullstack_save_tests tests/save_tests.cpp)
  target_link_libraries(nullstack_save_tests PRIVATE nullstack_save SQLite::SQLite3)
  add_test(NAME save COMMAND nullstack_save_tests)
  add_test(NAME headless_smoke COMMAND nullstack_headless --seed 42 --ticks 100000)
  add_test(NAME replay_cli COMMAND ${CMAKE_COMMAND} -DRUNNER=$<TARGET_FILE:nullstack_headless> -P ${CMAKE_SOURCE_DIR}/tests/integration/replay_cli.cmake)
endif()
