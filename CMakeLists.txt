cmake_minimum_required(VERSION 3.25)
project(NULLSTACK VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
add_library(nullstack_core STATIC src/core/simulation.cpp)
target_include_directories(nullstack_core PUBLIC src)
target_compile_features(nullstack_core PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(nullstack_core PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(nullstack_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(nullstack_headless src/headless/main.cpp)
target_link_libraries(nullstack_headless PRIVATE nullstack_core)
if(BUILD_TESTING)
  add_executable(nullstack_tests tests/core_tests.cpp)
  target_link_libraries(nullstack_tests PRIVATE nullstack_core)
  add_test(NAME core COMMAND nullstack_tests)
  add_test(NAME headless_smoke COMMAND nullstack_headless --seed 42 --ticks 100000)
endif()
